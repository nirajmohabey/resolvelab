def primitive(value):
    if isinstance(value, dict): return {key: primitive(item) for key, item in value.items()}
    if isinstance(value, (list, tuple)): return [primitive(item) for item in value]
    if isinstance(value, datetime): return value.astimezone(timezone.utc).isoformat()
    if isinstance(value, (str, int, float, bool)) or value is None: return value
    return str(value)

@contextmanager
def scoped(engine, tenant):
    with engine.begin() as conn:
        conn.execute(text("SELECT set_config('orderflow.tenant_id',:t,true),set_config('orderflow.actor_subject','internal-cloud-batch-verifier',true),set_config('orderflow.actor_role','pipeline',true),set_config('statement_timeout','15000',true),set_config('lock_timeout','3000',true)"), {"t":tenant})
        yield conn


def fixture_state(engine, tenant):
    with scoped(engine, tenant) as conn:
        counts = {name:conn.execute(text("SELECT count(*) FROM orderflow." + name)).scalar_one() for name in ("inbox_events","processing_receipts","entity_revisions","quarantines")}
        counts["inbox_statuses"] = {row[0]:row[1] for row in conn.execute(text("SELECT status,count(*) FROM orderflow.inbox_events GROUP BY status"))}
        return counts


def fixture_hashes(engine):
    result = {}
    for tenant in (ALPHA, BETA):
        with scoped(engine, tenant) as conn:
            result[tenant] = {}
            for table in BUSINESS:
                digest = sha256(); count = 0
                for value, in conn.execute(text('SELECT to_jsonb(t)::text FROM orderflow.' + table + ' t ORDER BY to_jsonb(t)::text COLLATE "C"')):
                    raw = value.encode(); digest.update(len(raw).to_bytes(8,"big")); digest.update(raw); count += 1
                result[tenant][table] = {"rows":count,"sha256":digest.hexdigest()}
    return result


def pointer(engine):

def canonical(payload):
    # Compile only the exact pure helper, without importing SDK/runtime settings.
    source = (ROOT / "src/orderflow/hosted_jobs.py").read_text()
    function = next(node for node in ast.parse(source).body if isinstance(node,ast.FunctionDef) and node.name == "_payload")
    namespace = {"json":json}
    exec(compile(ast.Module(body=[function],type_ignores=[]),"reviewed-_payload","exec"),namespace)
    return namespace["_payload"](payload)


def identities():
    path = HERE / "job-identities.json"
    values = json.loads(path.read_text()) if path.exists() else {}
    for name in ("daily-good","daily-good-after-fix","daily-good-after-runtime-fix","daily-wrong-oracle","inventory-mismatch"):
        if name not in values:
            key = str(uuid4())
            values[name] = {"idempotency_key":key,"job_id":str(uuid5(NAMESPACE_URL,"orderflow:"+ALPHA+":"+key))}
    path.write_text(json.dumps(values,indent=2)+"\n")
    return values


def enqueue(engine, name, kind, payload, identity):
    body = canonical(payload)
    assert len(body.encode()) <= 65536
    digest = sha256((kind+":"+body).encode()).hexdigest()
    job = identity["job_id"]
    with scoped(engine, ALPHA) as conn:
        conn.execute(text("SELECT pg_advisory_xact_lock(hashtextextended(:key,0))"),{"key":ALPHA+":job-budget"})
        rows = conn.execute(text("SELECT job_id::text,kind,payload_sha256 FROM orderflow.hosted_jobs WHERE kind!='drain'")).all()
        own_ids = {value["job_id"] for value in identities().values()}
        assert len(rows) <= 5 and all(row[0] in own_ids for row in rows), "Unexpected pre-existing batch job; stop for coordination"
        old = next((row for row in rows if row[0] == job), None)
        if old:
            assert old[1:] == (kind,digest), "Existing synthetic job identity differs"
        else:
            assert len(rows) < 5, "Authorized five-job bound reached"
            conn.execute(text("INSERT INTO orderflow.hosted_jobs(tenant_id,job_id,kind,payload,payload_sha256) VALUES(:t,:j,:kind,CAST(:p AS jsonb),:h)"),{"t":ALPHA,"j":job,"kind":kind,"p":body,"h":digest})
    record = {"name":name,"job_id":job,"kind":kind,"payload_sha256":digest,"canonical_helper_source_sha256":sha256((ROOT/"src/orderflow/hosted_jobs.py").read_bytes()).hexdigest(),"states":[]}
    PROOF["jobs"].append(record); save()
    return record


def collect(engine, record):
    job = record["job_id"]
    with scoped(engine, ALPHA) as conn:
        row = conn.execute(text("SELECT job_id,kind,phase,status,generation,attempts,available_at,published_generation,published_at,result,last_error_code,created_at,updated_at,lease_token IS NOT NULL AS lease_present FROM orderflow.hosted_jobs WHERE job_id=:j"),{"j":job}).mappings().one()
        ingestion = conn.execute(text("SELECT run_id,status,failure_code,started_at,finished_at FROM orderflow.ingestion_runs WHERE run_id=:j"),{"j":job}).mappings().one_or_none()
        candidate = conn.execute(text("SELECT status,generation,snapshot_sha256,oracle_sha256,project_sha256,manifest_sha256,report_payload FROM orderflow.report_candidates WHERE run_id=:j"),{"j":job}).mappings().one_or_none()
        checks = conn.execute(text("SELECT check_name,status,observed,expected FROM orderflow.quality_results WHERE run_id=:j ORDER BY checked_at,check_id"),{"j":job}).mappings().all()
    return primitive(dict(row)), primitive(dict(ingestion)) if ingestion else None, primitive(dict(candidate)) if candidate else None, primitive([dict(v) for v in checks])


def wait(engine, record, secret):
    dispatch(secret); deadline=time.monotonic()+600; last_dispatch=time.monotonic()
    while time.monotonic() < deadline:
        row, run, candidate, checks = collect(engine, record)
        marker = {key:row[key] for key in ("phase","status","generation","attempts","published_generation","last_error_code","lease_present")}
        if not record["states"] or marker != record["states"][-1]["state"]:
            record["states"].append({"at":datetime.now(timezone.utc).isoformat(),"state":marker})
            print(json.dumps({"job":record["name"],"state":marker}),flush=True)
        record.update(final=row,ingestion_run=run,candidate=candidate,checks=checks); save()
        if row["status"] in {"succeeded","failed"}: return
        if row["status"] == "pending" and row["published_generation"] != row["generation"] and time.monotonic()-last_dispatch>20:
            dispatch(secret); last_dispatch=time.monotonic()
        time.sleep(3)
    raise RuntimeError("Synthetic hosted job exceeded bounded observer deadline")


def objects(engine, record):
    with scoped(engine, ALPHA) as conn:
        pattern = "%"+record["job_id"]+"%" if record["kind"] == "daily" else "%/"+CSV+"/INV-001/%"
        rows = conn.execute(text("SELECT object_key,sha256,content,pg_typeof(content)::text AS type FROM orderflow.stored_objects WHERE object_key LIKE :p ORDER BY object_key"),{"p":pattern}).all()
    result=[]
    for key,expected,content,kind in rows:
        content=bytes(content); digest=sha256(content).hexdigest()
        assert digest == expected and kind == "bytea", "Cloud object checksum/type mismatch"
        entry={"key":key,"sha256":digest,"bytes":len(content),"type":kind,"checksum_verified":True}
        if "/dbt-" in key and key.endswith("results.json"):
            data=json.loads(content)
            entry["dbt"]={"version":data.get("metadata",{}).get("dbt_version"),"results":[{k:item.get(k) for k in ("unique_id","status","failures","execution_time")} for item in data.get("results",[])]}
        result.append(entry)
    record["objects"]=result
    with scoped(engine,BETA) as conn:
        counts={table:conn.execute(text("SELECT count(*) FROM orderflow."+table+" WHERE tenant_id=:alpha"),{"alpha":ALPHA}).scalar_one() for table in ("hosted_jobs","stored_objects","report_candidates","report_publications","inventory_snapshots")}
    assert not any(counts.values()), "Beta context observed Alpha pipeline data"
    record["beta_read_only_rls_denial"]={"counts":counts,"all_zero":True,"scope":"Restricted-app SQL tenant context, not an enrolled Beta HTTP/JWT principal."}; save()
